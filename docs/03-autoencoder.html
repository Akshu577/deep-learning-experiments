# Preprocess the data


def preprocess(imgs):
    """
    Normalize and reshape the images
    """
    imgs = imgs.astype("float32") / 255.0
    imgs = np.pad(imgs, ((0, 0), (2, 2), (2, 2)), constant_values=0.0)
    imgs = np.expand_dims(imgs, -1)
    return imgs

# Encoder
encoder_input = layers.Input(
    shape=(IMAGE_SIZE, IMAGE_SIZE, CHANNELS), name="encoder_input"
)
x = layers.Conv2D(32, (3, 3), strides=2, activation="relu", padding="same")(
    encoder_input
)
x = layers.Conv2D(64, (3, 3), strides=2, activation="relu", padding="same")(x)
x = layers.Conv2D(128, (3, 3), strides=2, activation="relu", padding="same")(x)
shape_before_flattening = K.int_shape(x)[1:]  # the decoder will need this!

x = layers.Flatten()(x)
encoder_output = layers.Dense(EMBEDDING_DIM, name="encoder_output")(x)

encoder = models.Model(encoder_input, encoder_output)

# Decoder
decoder_input = layers.Input(shape=(EMBEDDING_DIM,), name="decoder_input")
x = layers.Dense(np.prod(shape_before_flattening))(decoder_input)
x = layers.Reshape(shape_before_flattening)(x)
x = layers.Conv2DTranspose(
    128, (3, 3), strides=2, activation="relu", padding="same"
)(x)
x = layers.Conv2DTranspose(
    64, (3, 3), strides=2, activation="relu", padding="same"
)(x)
x = layers.Conv2DTranspose(
    32, (3, 3), strides=2, activation="relu", padding="same"
)(x)
decoder_output = layers.Conv2D(
    CHANNELS,
    (3, 3),
    strides=1,
    activation="sigmoid",
    padding="same",
    name="decoder_output",
)(x)

decoder = models.Model(decoder_input, decoder_output)

6. Discrepancies between original and reconstructed images¶

I picked five images that cover very different clothing shapes: an ankle boot, a pair of trousers, a coat, a sandal, and a bag. Because the encoder compresses each 32x32 image down to just two numbers (EMBEDDING_DIM = 2), the decoder only has a tiny amount of information to work with when it rebuilds the image, so some loss of detail is expected everywhere. Looking at the actual outputs:

  • Sandal (MSE = 0.0107) and trousers (MSE = 0.0113) reconstructed the best out of the five. Both have simple, blocky silhouettes with large uniform regions, so the decoder was able to reproduce their outline almost cleanly, even though the edges came out slightly softened compared to the sharp original.
  • Ankle boot (MSE = 0.0154) kept its overall shoe shape and sole, but the reconstruction smudges the finer structure, like the ankle collar and the boundary between the sole and upper, into a single blurry mass. The item is still clearly recognizable as footwear, though.
  • Bag (MSE = 0.0202) kept its rectangular body, but the reconstruction lost most of the thin shoulder strap and the fine seam/clasp lines visible in the original, turning the bag into a mostly solid dark block.
  • Coat (MSE = 0.0269), which had the highest error of the five, showed the most noticeable discrepancy. The original has a light body with visible dark zipper trim and clearly separated sleeves, but the reconstruction blurs all of that into a plain grey pullover-like shape, losing the trim contrast and even part of one sleeve boundary.

The general pattern is that items with a simple, uniform silhouette (sandal, trousers) reconstruct almost cleanly, while items with fine internal detail or thin structures (straps, zippers, sharply contrasting trim) lose that detail and get "averaged out" into a smoother, blobbier version of the original. This makes sense given the extremely small 2D latent bottleneck: the network can only preserve the coarsest, most dominant shape information, not fine texture or edges, which is why every reconstruction looks like a blurred version of the original rather than a sharp copy.
